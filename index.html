// --- MODULO: CALCOLO INTERESSI LEGALI ---


const InteressiLegaliCalculator = () => {

  const [capitale, setCapitale] = useState(1000);

  const [dataInizio, setDataInizio] = useState('2023-01-01');

  const [dataFine, setDataFine] = useState(new Date().toISOString().split('T')[0]);

  const [anatocismo, setAnatocismo] = useState(false);

  const [risultato, setRisultato] = useState(null);


  // Esempio semplificato di tassi legali storici

  const tassiLegali = [

    { dal: '2022-01-01', al: '2022-12-31', tasso: 1.25 },

    { dal: '2023-01-01', al: '2023-12-31', tasso: 5.0 },

    { dal: '2024-01-01', al: '2024-12-31', tasso: 2.5 },

    { dal: '2025-01-01', al: '2025-12-31', tasso: 2.0 },

  ];


  const calcolaInteressi = () => {

    const inizio = new Date(dataInizio);

    const fine = new Date(dataFine);

    

    if (fine <= inizio) return;


    const diffTime = Math.abs(fine - inizio);

    const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

    

    // Calcolo semplificato (usando l'ultimo tasso disponibile per brevità dell'esempio)

    // In un'app reale si itererebbe sui periodi storici dei tassi

    const tassoAttuale = 2.5; // Tasso ipotetico medio

    const interessi = (capitale * tassoAttuale * diffDays) / (365 * 100);

    const totale = capitale + interessi;


    setRisultato({

      interessi,

      totale,

      giorni: diffDays,

      tassoApplicato: tassoAttuale

    });

  };


  useEffect(() => {

    calcolaInteressi();

  }, [capitale, dataInizio, dataFine, anatocismo]);


  return (

    <div className="p-5 space-y-4 animate-in slide-in-from-right-4 duration-300">

      <div className="bg-white rounded-2xl shadow-lg border p-6 space-y-4">

        <div>

          <label className="block text-sm font-semibold text-slate-700 mb-1">Capitale Iniziale</label>

          <div className="relative">

            <span className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400 font-bold">€</span>

            <input 

              type="number" 

              value={capitale}

              onChange={(e) => setCapitale(parseFloat(e.target.value) || 0)}

              className="w-full p-3 pl-8 border rounded-xl focus:ring-2 focus:ring-sky-500 outline-none bg-slate-50 font-bold"

            />

          </div>

        </div>


        <div className="grid grid-cols-2 gap-4">

          <div>

            <label className="block text-xs font-semibold text-slate-500 mb-1">Dalla data</label>

            <input 

              type="date" 

              value={dataInizio}

              onChange={(e) => setDataInizio(e.target.value)}

              className="w-full p-2 border rounded-lg text-sm bg-slate-50"

            />

          </div>

          <div>

            <label className="block text-xs font-semibold text-slate-500 mb-1">Alla data</label>

            <input 

              type="date" 

              value={dataFine}

          </div>

            type="checkbox"

            Capitalizzazione (Anatocismo)

// --- MODULO: CALCOLO TERMINI PROCESSUALI ---


const TerminiProcessor = () => {

  const [startDate, setStartDate] = useState(new Date().toISOString().split('T')[0]);

  const [days, setDays] = useState(30);

  const [includeSuspension, setIncludeSuspension] = useState(true);

  const [resultDate, setResultDate] = useState(null);


  const calculateDeadline = () => {

    let date = new Date(startDate);

    let daysToAdd = parseInt(days);

    

    if (isNaN(daysToAdd)) return;

    date.setDate(date.getDate() + daysToAdd);


    if (includeSuspension) {

      const startYear = new Date(startDate).getFullYear();

      const augStart = new Date(startYear, 7, 1);

      const augEnd = new Date(startYear, 7, 31);

      if (new Date(startDate) <= augEnd && date >= augStart) {

        date.setDate(date.getDate() + 31);

      }

    }


    const dayOfWeek = date.getDay();

    if (dayOfWeek === 6) date.setDate(date.getDate() + 2);

    else if (dayOfWeek === 0) date.setDate(date.getDate() + 1);


    setResultDate(date);

  };


  useEffect(() => {

    calculateDeadline();

  }, [startDate, days, includeSuspension]);


  return (

    <div className="p-5 animate-in fade-in slide-in-from-bottom-4 duration-300">

      <div className="bg-white rounded-2xl shadow-lg border p-6 space-y-6">

        <div>

          <label className="block text-sm font-semibold text-slate-700 mb-2">Data di decorrenza</label>

          <input type="date" value={startDate} onChange={(e) => setStartDate(e.target.value)} className="w-full p-3 border rounded-xl outline-none bg-slate-50" />

        </div>

        <div>

          <label className="block text-sm font-semibold text-slate-700 mb-2">Termine (in giorni)</label>

          <input type="number" value={days} onChange={(e) => setDays(e.target.value)} className="w-full p-3 border rounded-xl outline-none bg-slate-50" />

        </div>

        <div className="flex items-center gap-3 p-3 bg-blue-50 rounded-xl border border-blue-100">

          <input type="checkbox" id="suspension" checked={includeSuspension} onChange={(e) => setIncludeSuspension(e.target.checked)} className="w-5 h-5 rounded accent-blue-600" />

          <label htmlFor="suspension" className="text-sm font-medium text-blue-800">Includi Sospensione Feriale</label>

        </div>

        {resultDate && (

          <div className="mt-8 p-6 bg-slate-900 text-white rounded-2xl text-center shadow-inner">

            <span className="text-slate-400 text-xs uppercase font-bold tracking-widest">Scadenza</span>

            <div className="text-3xl font-black mt-2">{resultDate.toLocaleDateString('it-IT', { day: '2-digit', month: 'long', year: 'numeric' })}</div>

          </div>

        )}

      </div>

    </div>

  );

};


// --- MODULO: FATTURAZIONE AVVOCATI ---


const FatturazioneAvvocato = () => {

  const [taxable, setTaxable] = useState(1000);

  const [cpaRate, setCpaRate] = useState(4);

  const [vatRate, setVatRate] = useState(22);

  const [applyRitenuta, setApplyRitenuta] = useState(true);


  const cpaAmount = (taxable * cpaRate) / 100;

  const taxableTotal = taxable + cpaAmount;

  const vatAmount = (taxableTotal * vatRate) / 100;

  const grandTotal = taxableTotal + vatAmount;

  const ritenutaAmount = applyRitenuta ? (taxable * 20) / 100 : 0;

  const netPayable = grandTotal - ritenutaAmount;


  const formatEuro = (val) => new Intl.NumberFormat('it-IT', { style: 'currency', currency: 'EUR' }).format(val);


  return (

    <div className="p-5 animate-in fade-in slide-in-from-bottom-4 duration-300">

      <div className="bg-white rounded-2xl shadow-lg border p-6">

        <div className="space-y-4 mb-6">

          <div>

            <label className="block text-sm font-semibold text-slate-700 mb-1">Onorari</label>

            <input type="number" value={taxable} onChange={(e) => setTaxable(parseFloat(e.target.value) || 0)} className="w-full p-3 border rounded-xl bg-slate-50 font-bold" />

          </div>

          <div className="grid grid-cols-2 gap-4">

            <input type="number" value={cpaRate} onChange={(e) => setCpaRate(e.target.value)} className="p-2 border rounded-lg text-sm bg-slate-50" placeholder="CPA %" />

            <input type="number" value={vatRate} onChange={(e) => setVatRate(e.target.value)} className="p-2 border rounded-lg text-sm bg-slate-50" placeholder="IVA %" />

          </div>

          <div className="flex items-center justify-between p-3 bg-slate-50 rounded-xl border">

            <span className="text-sm font-medium">Ritenuta 20%</span>

            <input type="checkbox" checked={applyRitenuta} onChange={(e) => setApplyRitenuta(e.target.checked)} className="w-6 h-6 accent-green-600" />

          </div>

        </div>

        <div className="mt-4 p-4 bg-green-600 text-white rounded-xl text-center">

          <span className="text-xs uppercase font-bold">Netto a Pagare</span>

          <div className="text-3xl font-black">{formatEuro(netPayable)}</div>

        </div>

      </div>

    </div>

  );

};
